int main()
{
    int a[10][10], b[10][10], c[10][10];
    int r1, c1, r2, c2, i, j, k;
    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1, &c1);
    printf("Enter rows and columns of second matrix: ");
    scanf("%d %d", &r2, &c2);
    if(c1 != r2)
    {
        printf("Matrix multiplication not possible");
        return 0;
    }
    printf("Enter first matrix:\n");
    for(i = 0; i < r1; i++)
        for(j = 0; j < c1; j++)
            scanf("%d", &a[i][j]);
    printf("Enter second matrix:\n");
    for(i = 0; i < r2; i++)
        for(j = 0; j < c2; j++)
            scanf("%d", &b[i][j]);
    for(i = 0; i < r1; i++)
        for(j = 0; j < c2; j++)
        {
            c[i][j] = 0;
            for(k = 0; k < c1; k++)
                c[i][j] += a[i][k] * b[k][j];
        }
    printf("Result matrix:\n");
    for(i = 0; i < r1; i++)
    {
        for(j = 0; j < c2; j++)
            printf("%d ", c[i][j]);
        printf("\n");
    }

    return 0;
}

long long factorial(int n)
{
    if(n == 0 || n == 1)
        return 1;
    return n * factorial(n - 1);
}

19)HEAP SORT
CODE:
#include <stdio.h>
void heapify(int a[], int n, int i)
{
    int largest, left, right, temp;
    largest = i;
    left = 2 * i + 1;
    right = 2 * i + 2;
    if(left < n && a[left] > a[largest])
        largest = left;
    if(right < n && a[right] > a[largest])
        largest = right;

    if(largest != i)
    {
        temp = a[i];
        a[i] = a[largest];
        a[largest] = temp;

        heapify(a, n, largest);
    }
}

24)  Minimum Spanning Tree – Prim's Algorithm
CODE:
 #include <stdio.h>
#define INF 999

int main()
{
    int cost[10][10], selected[10] = {0};
    int n, i, j, edges = 0;
    int min, x, y, total = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter cost matrix:\n");

    for(i = 0; i < n; i++)
        for(j = 0; j < n; j++)
            scanf("%d", &cost[i][j]);

    selected[0] = 1;

    printf("Edges in MST:\n");

    while(edges < n - 1)
    {
        min = INF;
        x = y = -1;

        for(i = 0; i < n; i++)
        {
            if(selected[i])
            {
                for(j = 0; j < n; j++)
                {
                    if(!selected[j] && cost[i][j] < min)
                    {
                        min = cost[i][j];
                        x = i;
                        y = j;
                    }
                }
            }
        }

        printf("%d - %d = %d\n", x, y, min);

        total += min;
        selected[y] = 1;
        edges++;
    }

    printf("Minimum Cost = %d\n", total);

    return 0;
}


25)Minimum Spanning Tree – Kruskal's Algorithm
Code#include <stdio.h>

struct Edge
{
    int u, v, w;
};

int parent[10];

int find(int x)
{
    while(parent[x] != x)
        x = parent[x];

    return x;
}

int main()
{
    struct Edge e[20], temp;
    int n, m, i, j, a, b;
    int count = 0, total = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter number of edges: ");
    scanf("%d", &m);

    printf("Enter edges (source destination weight):\n");

    for(i = 0; i < m; i++)
        scanf("%d %d %d", &e[i].u, &e[i].v, &e[i].w);

    for(i = 0; i < n; i++)
        parent[i] = i;

    for(i = 0; i < m - 1; i++)
    {
        for(j = i + 1; j < m; j++)
        {
            if(e[i].w > e[j].w)
            {
                temp = e[i];
                e[i] = e[j];
                e[j] = temp;
            }
        }
    }

    printf("Edges in MST:\n");

    for(i = 0; i < m && count < n - 1; i++)
    {
        a = find(e[i].u);
        b = find(e[i].v);

        if(a != b)
        {
            printf("%d - %d = %d\n",
                   e[i].u, e[i].v, e[i].w);

            total += e[i].w;
            parent[a] = b;
            count++;
        }
    }

    printf("Minimum Cost = %d\n", total);

    return 0;
}
